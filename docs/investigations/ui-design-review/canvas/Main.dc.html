/// Split ratio is owned by the model so the divider can drag it.
private struct ProjectPaneLayoutView: View {
@ObservedObject var surface: ProjectSurfaceModel
let node: ProjectPaneNode
private func splitView(
_ orientation: ProjectPaneOrientation,
ratio: Double,
first: ProjectPaneNode,
second: ProjectPaneNode
) -> some View {
GeometryReader { proxy in
let span = orientation == .horizontal
? proxy.size.width
: proxy.size.height
PaneSplit(orientation: orientation, ratio: ratio) {
nodeView(first)
} second: {
nodeView(second)
} onDrag: { translation in
surface.updateSplitRatio(
nodeID: node.id,
ratio: ratio + translation / span
)
}
}
}
/// The 1px divider owns a ±4px hit area and a resize cursor.
private var divider: some View {
Rectangle()
.fill(WorkspaceChrome.border)
.frame(width: 1)
.contentShape(Rectangle().inset(by: -4))
.onHover { hovering in
isHovering = hovering
hovering
? NSCursor.resizeLeftRight.push()
: NSCursor.pop()
}
}
}
